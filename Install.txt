Connect Rpi via USB Mini Cable to Laptop


#INSTALL

sudo apt install i2c-tools python3-smbus hostapd dnsmasq python3-pip git cmake mosquitto mosquitto-clients -y
pip install setuptools paho-mqtt mpu6050-raspberrypi pyserial --break-system-packages

sudo systemctl stop dnsmasq hostapd

sudo nano /etc/dhcpcd.conf

# add
    interface wlan0
        static ip_address=192.168.4.1/24
        nohook wpa_supplicant



sudo rm /etc/NetworkManager/system-connections/*.nmconnection
sudo rm /etc/wpa_supplicant/wpa_supplicant.conf

sudo nano /etc/hostapd/hostapd.conf

# add
    interface=wlan0
    driver=nl80211
    ssid=PiZero_AP
    hw_mode=g
    channel=7
    auth_algs=1
    ignore_broadcast_ssid=0


sudo nano /etc/default/hostapd
# change
    DAEMON_CONF="/etc/hostapd/hostapd.conf"



sudo mv /etc/dnsmasq.conf /etc/dnsmasq.conf.orig
sudo nano /etc/dnsmasq.conf

# add 
    interface=wlan0
    dhcp-range=192.168.4.2,192.168.4.20,255.255.255.0,24h


sudo systemctl unmask hostapd
sudo systemctl enable hostapd
sudo systemctl enable dnsmasq

sudo systemctl restart dnsmasq
sudo systemctl restart hostapd



✅ Make NetworkManager ignore wlan0 (Bookworm)

Create a config file:

sudo mkdir -p /etc/NetworkManager/conf.d
sudo nano /etc/NetworkManager/conf.d/wlan0-unmanaged.conf


Add:

[keyfile]
unmanaged-devices=interface-name:wlan0


Save → exit → then reload NM:

sudo systemctl restart NetworkManager


Create:

sudo nano /etc/systemd/network/08-wlan0.network


Insert:

[Match]
Name=wlan0

[Network]
Address=192.168.4.1/24
DHCPServer=no


Apply:

sudo systemctl restart systemd-networkd
